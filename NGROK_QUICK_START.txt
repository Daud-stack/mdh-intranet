✅ Step 1: ngrok is Downloaded!
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   Location: c:\Users\HomePC\mdh_intranet\ngrok\ngrok.exe
   Version: 3.35.0

🔑 Step 2: Get FREE ngrok Authentication Token
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   1. Go to: https://dashboard.ngrok.com/signup
   2. Sign up (FREE - use Google/GitHub/Email)
   3. Go to: https://dashboard.ngrok.com/get-started/your-authtoken
   4. Copy your authtoken

⚙️  Step 3: Configure ngrok
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   Open PowerShell in: c:\Users\HomePC\mdh_intranet
   
   Run this command (replace YOUR_TOKEN with your actual token):
   
   .\ngrok\ngrok.exe config add-authtoken YOUR_TOKEN
   
   Example:
   .\ngrok\ngrok.exe config add-authtoken 2a1b2c3d4e5f6g7h8i9j

🚀 Step 4: Start ngrok Tunnel
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   Run this command:
   
   .\ngrok\ngrok.exe http 8000
   
   You'll see output like:
   
   ┌───────────────────────────────────────────────────┐
   │ Session Status   online                           │
   │ Forwarding       https://abc123.ngrok.io ->       │
   │                  http://localhost:8000            │
   └───────────────────────────────────────────────────┘
   
   📝 COPY THIS URL: https://abc123.ngrok.io
   
   ⚠️  KEEP THIS WINDOW OPEN! Don't close it!

🔧 Step 5: Update Django Settings
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   OPTION A - Automatic (Easy):
   
   Open a NEW PowerShell window and run:
   
   .\update_ngrok_host.ps1 https://abc123.ngrok.io
   
   (Replace abc123.ngrok.io with YOUR actual URL)
   
   ─────────────────────────────────────────────────────
   
   OPTION B - Manual:
   
   1. Edit: mdh_intranet\settings.py
   2. Find the line with ALLOWED_HOSTS
   3. Add your ngrok URL (without https://):
   
   ALLOWED_HOSTS = os.environ.get(
       'DJANGO_ALLOWED_HOSTS',
       'localhost,127.0.0.1,[::1],host.docker.internal,10.232.190.161,abc123.ngrok.io'
   ).split(',')

                                                                              Add your URL here

🔄 Step 6: Restart Django Server
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   1. In the Django terminal, press: Ctrl+C
   2. Run: python manage.py runserver
   3. Wait for "Starting development server at http://127.0.0.1:8000/"

🌐 Step 7: Access Your App via ngrok
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   1. Open your browser
   2. Go to: https://abc123.ngrok.io (use YOUR URL)
   3. Login to MDH Intranet
   4. You should see your app!

📄 Step 8: Test Office Web Viewer
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   1. Navigate to: /documents/
   2. Upload a .docx, .xlsx, or .pptx file
   3. Click "👁 View Online" button
   4. The document should open in Office Web Viewer!

⚠️  Important Notes
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   ⭐ ALWAYS access via the ngrok URL (https://abc123.ngrok.io)
      NOT via localhost:8000!

   ⭐ KEEP the ngrok terminal window OPEN
      If you close it, the tunnel stops

   ⭐ FREE ngrok URLs change each restart
      You'll need to update ALLOWED_HOSTS each time

   ⭐ Your app is now PUBLIC on the internet
      Anyone with the URL can access it (while ngrok is running)

🐛 Troubleshooting
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   Problem: "Invalid authtoken"
   → Make sure you copied the complete token from ngrok dashboard

   Problem: Office Viewer shows error
   → Make sure you're accessing via ngrok URL, not localhost
   → Verify ALLOWED_HOSTS includes your ngrok URL

   Problem: "This site can't be reached"
   → Make sure ngrok is still running
   → Check if Django server is running

   Problem: ngrok window shows errors
   → Read the error message carefully
   → Most common: Need to add authtoken (Step 3)
